# plugin CMake file to be included in main project CMakeLists

# --- IA Assistant Plugin ---
# Plugin for AI-assisted table creation and editing

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_C_STANDARD 99)

set(BUILD_LIBS_DIR "${CMAKE_SOURCE_DIR}/third-party/build-libs/${PLATFORM}-${ARCH}")
set(RUNTIME_LIBS_DIR "${CMAKE_SOURCE_DIR}/third-party/runtime-libs/${PLATFORM}-${ARCH}")

set(IA_PLUGIN_SOURCES
   plugins/iaassistant/ia_assistant.cpp
   plugins/iaassistant/ai_client.cpp
   plugins/iaassistant/table_analyzer.cpp
)

if(BUILD_SHARED)
   add_library(IAAssistantPlugin MODULE ${IA_PLUGIN_SOURCES})

   target_include_directories(IAAssistantPlugin PUBLIC
      ${CMAKE_SOURCE_DIR}/third-party/include
      plugins
      plugins/iaassistant
   )

   if(PLATFORM STREQUAL "windows" OR PLATFORM STREQUAL "windows-mingw")
      if(MSVC)
         set_target_properties(IAAssistantPlugin PROPERTIES
            MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>"
         )
         set(OPT_COMMON
            /Ob2
            /O2
            /Oi
            /fp:fast
            /fp:except-
            /Ot
            /GF
            /GS-
            /Gy
            /GR
            /Oy
            /GT
            /GL
            /w44005
            /Zc:__cplusplus
            /std:c++20
            $<$<NOT:$<STREQUAL:${ARCH},x64>>:/arch:SSE2>
         )
         target_compile_options(IAAssistantPlugin PUBLIC
            $<$<CONFIG:Release>:$<$<COMPILE_LANGUAGE:CXX>:${OPT_COMMON}>>
            $<$<CONFIG:Release>:$<$<COMPILE_LANGUAGE:C>:${OPT_COMMON}>>
         )
         target_link_options(IAAssistantPlugin PUBLIC
            $<$<CONFIG:Release>:/INCREMENTAL:NO>
            $<$<CONFIG:Release>:/OPT:REF>
            $<$<CONFIG:Release>:/OPT:ICF>
            $<$<CONFIG:Release>:/LTCG>
         )
      endif()
      target_link_directories(IAAssistantPlugin PUBLIC ${BUILD_LIBS_DIR} ${RUNTIME_LIBS_DIR})
      set_target_properties(IAAssistantPlugin PROPERTIES
         PREFIX ""
         OUTPUT_NAME "$<IF:$<STREQUAL:${ARCH},x64>,plugin-iaassistant64,plugin-iaassistant>"
      )
   else()
      set_target_properties(IAAssistantPlugin PROPERTIES
         PREFIX ""
         OUTPUT_NAME "plugin-iaassistant"
      )
      if(PLATFORM STREQUAL "macos")
         set_target_properties(IAAssistantPlugin PROPERTIES
            SUFFIX ".dylib"
         )
      endif()
      target_link_directories(IAAssistantPlugin PUBLIC ${RUNTIME_LIBS_DIR})
   endif()

   set_target_properties(IAAssistantPlugin PROPERTIES
      LIBRARY_OUTPUT_DIRECTORY "${PLUGINS_DIR}/iaassistant"
   )

   add_custom_command(TARGET IAAssistantPlugin POST_BUILD
      COMMAND "${CMAKE_COMMAND}" -E copy "${CMAKE_SOURCE_DIR}/plugins/iaassistant/plugin.cfg" "${PLUGINS_DIR}/iaassistant/plugin.cfg"
   )

   if(POST_BUILD_COPY_EXT_LIBS)
      if(PLATFORM STREQUAL "linux")
         add_custom_command(TARGET IAAssistantPlugin POST_BUILD
            COMMAND cp -a "${RUNTIME_LIBS_DIR}/libcurl.so" "${PLUGINS_DIR}/iaassistant/" 2>/dev/null || true
            COMMAND cp -a "${RUNTIME_LIBS_DIR}/libcurl.so."* "${PLUGINS_DIR}/iaassistant/" 2>/dev/null || true
         )
      endif()
   endif()
endif()

if(BUILD_STATIC)
   add_compile_options(-fPIC)

   add_library(IAAssistantPlugin STATIC ${IA_PLUGIN_SOURCES})

   target_include_directories(IAAssistantPlugin PUBLIC
      ${CMAKE_SOURCE_DIR}/third-party/include
      plugins
      plugins/iaassistant
   )

   set_target_properties(IAAssistantPlugin PROPERTIES
      PREFIX ""
      OUTPUT_NAME "plugin-iaassistant"
   )
endif()
